cat << 'EOF' > setup_vault_demo.sh
#!/bin/bash
set -e

echo "==> 1. Установка ansible-core..."
sudo apt update && sudo apt install -y ansible-core

echo "==> 2. Создание файла с паролем для Vault..."
echo "123" > .vault_pass
chmod 600 .vault_pass
echo ".vault_pass" >> .gitignore

echo "==> 3. Создание структуры каталогов и обычных переменных..."
mkdir -p group_vars/webservers

cat > group_vars/webservers/vars.yml << 'VARS_EOF'
db_host: "10.0.0.5"
db_port: 3306
db_user: "app_user"
VARS_EOF

echo "==> 4. Создание зашифрованного файла через CLI..."
# Используем автоматический ввод пароля из файла, чтобы скрипт не останавливался
ansible-vault encrypt_string 'ProductionPassword99!' --name 'vault_db_password' --vault-password-file .vault_pass > group_vars/webservers/vault.yml

echo "==> 5. Генерация инвентарного файла для локального теста (localhost)..."
cat > inventory.ini << 'INV_EOF'
[webservers]
localhost ansible_connection=local
INV_EOF

echo "==> 6. Создание плейбука с инлайн-зашифрованной строкой..."
# Генерируем зашифрованную строку во временный файл и собираем плейбук
ansible-vault encrypt_string 'SuperSecretPassword123!' --name 'db_password' --vault-password-file .vault_pass > /tmp/encrypted_var.txt

cat > test_vault.yml << 'PLAYBOOK_EOF'
---
- name: Test Inline Encrypted Variable
  hosts: webservers
  vars:
PLAYBOOK_EOF

# Добавляем отступы для переменной и дописываем задачи
sed 's/^/    /' /tmp/encrypted_var.txt >> test_vault.yml

cat >> test_vault.yml << 'PLAYBOOK_EOF'

  tasks:
    - name: Display unencrypted secret during runtime
      debug:
        msg: "The decrypted password is: {{ db_password }}"
PLAYBOOK_EOF

echo "==> 7. Запуск плейбука с использованием файла паролей..."
ansible-playbook -i inventory.ini test_vault.yml --vault-password-file .vault_pass

echo "==> Готово! Сценарий успешно выполнен."
EOF

chmod +x setup_vault_demo.sh
./setup_vault_demo.sh
